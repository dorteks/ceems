function cleanValue(value: unknown): string | null {
  if (value === undefined || value === null) {
    return null;
  }

  const str = String(value).trim();

  if (
    !str ||
    str.toLowerCase() === "null" ||
    str.toLowerCase() === "undefined" ||
    str === "." ||
    str.toLowerCase() === "<br>"
  ) {
    return null;
  }

  return str;
}

function toFloat(value: unknown): number | null {
  const cleaned = cleanValue(value);

  if (cleaned === null) {
    return null;
  }

  const parsed = Number(cleaned);

  return Number.isFinite(parsed) ? parsed : null;
}

// parse the xlsx
export async function importCustomersFromXlsx(
  fileBuffer: Buffer
) {
  const workbook = XLSX.read(fileBuffer, {
    type: "buffer",
    cellDates: true,
  });

  const sheetName = workbook.SheetNames[0];

  if (!sheetName) {
    throw new Error("The Excel file contains no worksheets.");
  }

  const worksheet = workbook.Sheets[sheetName];

  const rows = XLSX.utils.sheet_to_json<Record<string, unknown>>(
    worksheet,
    {
      defval: null,
    }
  );

  if (!rows.length) {
    throw new Error("The Excel worksheet contains no customer records.");
  }

  const customers = rows.map(mapCustomerRow);

  return customers;
}


// validate fields
function validateCustomer(
  customer: Record<string, any>,
  rowNumber: number
) {
  const errors: string[] = [];

  if (!customer.account_number) {
    errors.push("account_number is required");
  }

  if (!customer.surname) {
    errors.push("surname is required");
  }

  if (!customer.first_name) {
    errors.push("first_name is required");
  }

  if (errors.length) {
    return {
      valid: false,
      errors,
      row: rowNumber,
    };
  }

  return {
    valid: true,
    errors: [],
    row: rowNumber,
  };
}

function normalizeEnum(value: unknown): string | null {
  const cleaned = cleanValue(value);

  if (!cleaned) {
    return null;
  }

  return cleaned
    .toUpperCase()
    .trim()
    .replace(/&/g, "AND")
    .replace(/[\/\-]+/g, "_")
    .replace(/\s+/g, "_")
    .replace(/_+/g, "_");
}

// If you're using Express, your endpoint could look like:
import express from "express";
import multer from "multer";

const router = express.Router();

const upload = multer({
  storage: multer.memoryStorage(),
  limits: {
    fileSize: 20 * 1024 * 1024, // 20 MB
  },
});

router.post(
  "/customers/import",
  upload.single("file"),
  async (req, res) => {
    try {
      if (!req.file) {
        return res.status(400).json({
          message: "Please upload an XLSX file.",
        });
      }

      const result = await importCustomersFromXlsx(
        req.file.buffer
      );

      const validCustomers = [];
      const errors = [];

      for (let i = 0; i < result.length; i++) {
        const customer = result[i];

        const validation = validateCustomer(
          customer,
          i + 2 // +2 because Excel row 1 is the header
        );

        if (!validation.valid) {
          errors.push(validation);
          continue;
        }

        validCustomers.push(customer);
      }

      const BATCH_SIZE = 500;
      let imported = 0;

      for (
        let i = 0;
        i < validCustomers.length;
        i += BATCH_SIZE
      ) {
        const batch = validCustomers.slice(
          i,
          i + BATCH_SIZE
        );

        const created = await prisma.customer.createMany({
          data: batch,
          skipDuplicates: true,
        });

        imported += created.count;
      }

      return res.status(201).json({
        message: "Customer import completed.",
        total_rows: result.length,
        imported,
        failed: errors.length,
        errors,
      });
    } catch (error) {
      console.error("Customer import failed:", error);

      return res.status(500).json({
        message: "Failed to import customers.",
      });
    }
  }
);

export default router;
